## 2. 表结构与索引（ngram 全文 + 两个结构化索引）
*************************** 1. row ***************************
       Table: fusions
Create Table: CREATE TABLE `fusions` (
  `id` varchar(64) NOT NULL,
  `item_a` varchar(40) NOT NULL,
  `item_b` varchar(40) NOT NULL,
  `name` varchar(80) NOT NULL,
  `effect` varchar(400) NOT NULL,
  `side_effect` varchar(400) NOT NULL DEFAULT '',
  `rarity` varchar(10) NOT NULL DEFAULT '普通',
  `creator` varchar(40) NOT NULL DEFAULT '匿名发明家',
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FULLTEXT KEY `ft_content` (`name`, `effect`) WITH PARSER ngram PARSER_PROPERTIES=(ngram_token_size=2) BLOCK_SIZE 16384,
  KEY `idx_a` (`item_a`) BLOCK_SIZE 16384 LOCAL,
  KEY `idx_b` (`item_b`) BLOCK_SIZE 16384 LOCAL
) ORGANIZATION INDEX DEFAULT CHARSET = utf8mb4 ROW_FORMAT = DYNAMIC COMPRESSION = 'zstd_1.3.8' BLOCK_SIZE = 16384 TABLET_SIZE = 134217728 PCTFREE = 0
